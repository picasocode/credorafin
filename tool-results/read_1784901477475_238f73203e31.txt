    78→function parseCookie(): AdminUser|null {
    79→  // fallback client-side read (works only when httpOnly=false)
    80→  if (typeof document==="undefined") return null;
    81→  const m = document.cookie.match(/(?:^|;\s*)credora_admin_session=([^;]*)/);
    82→  if (!m) return null;
    83→  try { const d=JSON.parse(atob(decodeURIComponent(m[1]))); if(d.exp&&d.exp<Date.now()) return null; return d; }
    84→  catch { return null; }
    85→}

   103→function toCSV(rows:Record<string,unknown>[],name:string){
   104→  if(!rows.length)return;
   105→  const skip=new Set(["id","ip_address","user_agent","password_hash"]);
   106→  const h=Object.keys(rows[0]).filter(k=>!skip.has(k));
   107→  const csv=[h.join(","),...rows.map(r=>h.map(c=>`"${String(r[c]??"").replace(/"/g,'""')}"`).join(","))].join("\n");
   108→  const a=document.createElement("a");a.href=URL.createObjectURL(new Blob([csv],{type:"text/csv"}));
   109→  a.download=`${name}.csv`;a.click();
   110→}
