    49→/* ────────────────────────────────────────────

    52→interface CounterProps {
    53→  end: number;
    54→  suffix?: string;
    55→  prefix?: string;
    56→  duration?: number;
    57→  label: string;
    58→  sublabel?: string;

    60→
    61→function AnimatedCounter({
    62→  end,
    63→  suffix = "",
    64→  prefix = "",
    65→  duration = 2,
    66→  label,
    67→  sublabel,
    68→}: CounterProps) {
    69→  const [count, setCount] = useState(0);
    70→  const ref = useRef(null);
    71→  const isInView = useInView(ref, { once: true, margin: "-50px" });
    72→
    73→  useEffect(() => {
    74→    if (!isInView) return;
    75→    let start = 0;
    76→    const totalFrames = duration * 60;
    77→    const step = end / totalFrames;
    78→    // Ease-out cubic for smoother animation
    79→    const easeOut = (t: number) => 1 - Math.pow(1 - t, 3);
    80→    let frame = 0;
    81→    const timer = setInterval(() => {
    82→      frame++;
    83→      const progress = Math.min(frame / totalFrames, 1);
    84→      const easedProgress = easeOut(progress);
    85→      const currentValue = Math.floor(easedProgress * end);
    86→      if (progress >= 1) {
    87→        setCount(end);
    88→        clearInterval(timer);
    89→      } else {
    90→        setCount(currentValue);
    91→      }
    92→    }, 1000 / 60);
    93→    return () => clearInterval(timer);
    94→  }, [isInView, end, duration]);
    95→
    96→  return (
    97→    <motion.div
    98→      ref={ref}
    99→      className="text-center group"
   100→      initial={{ opacity: 0, y: 30 }}
   101→      animate={isInView ? { opacity: 1, y: 0 } : {}}
   102→      transition={{ duration: 0.6, type: "spring", stiffness: 100 }}
   103→    >
   104→      <motion.div
   105→        className="inline-block"
   106→        animate={isInView ? { scale: [1, 1.05, 1] } : {}}
   107→        transition={{ duration: 2, delay: 1, repeat: Infinity, ease: "easeInOut" }}
   108→      >
   109→        <div className="text-4xl sm:text-5xl lg:text-6xl font-bold text-white mb-2">
   110→          {prefix}
   111→          {count.toLocaleString()}
   112→          {suffix}
   113→        </div>
   114→      </motion.div>
   115→      <div className="text-base sm:text-lg text-white/90 font-medium">
   116→        {label}
   117→      </div>
   118→      {sublabel && (
   119→        <div className="text-sm text-white/60 mt-1">{sublabel}</div>
   120→      )}
   121→    </motion.div>
   122→  );
   123→}

   214→/* ── EMI calc helper (client-side, matches EMICalculator.tsx logic) ── */
   215→function calcEMI(principal: number, annualRate: number, tenureMonths: number): number {
   216→  if (principal <= 0 || annualRate <= 0 || tenureMonths <= 0) return 0;
   217→  const r = annualRate / 12 / 100;
   218→  return (principal * r * Math.pow(1 + r, tenureMonths)) / (Math.pow(1 + r, tenureMonths) - 1);
   219→}
   220→
   221→function formatINR(n: number): string {
   222→  if (!isFinite(n)) return "₹0";
   223→  if (n >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
   224→  if (n >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
   225→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   226→}
